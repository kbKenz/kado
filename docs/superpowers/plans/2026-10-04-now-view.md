```swift
import Foundation

/// Tracked time on one task or habit, as a value. `WorkSessionRecord`
/// stores it; this type does the arithmetic so views and tests never
/// repeat it.
nonisolated public struct WorkSession: Hashable, Sendable {
    public var startedAt: Date
    public var endedAt: Date?
    public var pausedAt: Date?
    /// Total of finished pauses. An open pause is counted from `pausedAt`.
    public var pausedSeconds: TimeInterval

    public init(startedAt: Date, endedAt: Date? = nil, pausedAt: Date? = nil, pausedSeconds: TimeInterval = 0) {
        self.startedAt = startedAt
        self.endedAt = endedAt
        self.pausedAt = pausedAt
        self.pausedSeconds = pausedSeconds
    }

    public var isOpen: Bool { endedAt == nil }
    public var isPaused: Bool { isOpen && pausedAt != nil }

    /// Worked time at `now`: (end or now) − start − pauses.
    public func elapsed(at now: Date) -> TimeInterval {
        let end = endedAt ?? now
        let openPause = isPaused ? max(0, end.timeIntervalSince(pausedAt!)) : 0
        return max(0, end.timeIntervalSince(startedAt) - pausedSeconds - openPause)
    }
}
```

```swift
import Foundation

/// What a block or session is about, as a value. Views address records
/// by these UUIDs and never hold a `@Model` (issue #63).
nonisolated public enum NowItem: Hashable, Identifiable, Sendable {
    case task(id: UUID, title: String)
    case habit(id: UUID, name: String)

    public var id: UUID {
        switch self {
        case .task(let id, _): return id
        case .habit(let id, _): return id
        }
    }

    public var title: String {
        switch self {
        case .task(_, let title): return title
        case .habit(_, let name): return name
        }
    }
}

/// A planned block already filtered to things that can be worked on.
nonisolated public struct NowBlock: Hashable, Sendable {
    public var id: UUID
    public var item: NowItem
    public var start: Date
    public var end: Date?
    public var createdAt: Date

    public init(id: UUID, item: NowItem, start: Date, end: Date?, createdAt: Date) {
        self.id = id
        self.item = item
        self.start = start
        self.end = end
        self.createdAt = createdAt
    }

    /// A block without an end lasts one hour (`ScheduleDefaults`).
    public var range: ClosedRange<Date> {
        start...max(start, end ?? start.addingTimeInterval(3600))
    }
}

/// The open session, if any, with what it is about.
nonisolated public struct OpenSession: Hashable, Sendable {
    public var id: UUID
    public var item: NowItem
    public var session: WorkSession
    public var blockID: UUID?

    public init(id: UUID, item: NowItem, session: WorkSession, blockID: UUID?) {
        self.id = id
        self.item = item
        self.session = session
        self.blockID = blockID
    }
}

nonisolated public enum NowState: Equatable, Sendable {
    case running(OpenSession, plannedRange: ClosedRange<Date>?)
    case paused(OpenSession, plannedRange: ClosedRange<Date>?)
    case suggestedCurrent(NowBlock)
    case suggestedNext(NowBlock)
    case empty
}

/// Everything the Now tab shows.
nonisolated public struct NowScreen: Equatable, Sendable {
    public var state: NowState
    public var upNext: NowBlock?

    public init(state: NowState, upNext: NowBlock?) {
        self.state = state
        self.upNext = upNext
    }
}
```

```swift
import Foundation

/// Decides what the Now tab shows. Pure: callers pass today's blocks
/// already filtered to workable items (`NowInputBuilder` in the app).
nonisolated public struct NowResolver: Sendable {
    public let boundary: DayBoundary

    public init(boundary: DayBoundary) {
        self.boundary = boundary
    }

    public func resolve(now: Date, blocks: [NowBlock], openSession: OpenSession?) -> NowScreen {
        let today = blocks
            .filter { boundary.isDate($0.start, inSameDayAs: now) }
            .sorted { ($0.start, $0.createdAt) < ($1.start, $1.createdAt) }

        if let openSession {
            let range = openSession.blockID.flatMap { id in blocks.first { $0.id == id }?.range }
            let anchor = range?.lowerBound ?? now
            let next = today.first { $0.start > anchor && $0.id != openSession.blockID }
            let state: NowState = openSession.session.isPaused
                ? .paused(openSession, plannedRange: range)
                : .running(openSession, plannedRange: range)
            return NowScreen(state: state, upNext: next)
        }

        if let current = today.first(where: { $0.range.contains(now) }) {
            return NowScreen(state: .suggestedCurrent(current), upNext: today.first { $0.start > current.start })
        }
        if let next = today.first(where: { $0.start > now }) {
            return NowScreen(state: .suggestedNext(next), upNext: nil)
        }
        return NowScreen(state: .empty, upNext: nil)
    }
}
```

Note: the `gap` test expects `upNext == nil` when the next block is already the main card; the code returns `nil` for `.suggestedNext`, so "Up next" never repeats the main card.

```swift
import Foundation
import SwiftData
import KadoCore

/// Every write to tracked time. Views call these with records they
/// resolved by UUID from their own `@Query`, then forget them.
@MainActor
struct WorkSessionTracker {
    enum TrackerError: Error, Equatable {
        case sessionAlreadyOpen
        case noOpenSession
    }

    let boundary: DayBoundary
    let now: () -> Date

    init(boundary: DayBoundary, now: @escaping () -> Date = { .now }) {
        self.boundary = boundary
        self.now = now
    }

    /// The open session, earliest first if sync brought in two.
    static func openSession(in context: ModelContext) throws -> WorkSessionRecord? {
        var descriptor = FetchDescriptor<WorkSessionRecord>(
            predicate: #Predicate { $0.endedAt == nil },
            sortBy: [SortDescriptor(\.startedAt)]
        )
        descriptor.fetchLimit = 1
        return try context.fetch(descriptor).first
    }

    func start(task: TaskRecord, block: ScheduleBlockRecord?, in context: ModelContext) throws {
        try insert(WorkSessionRecord(startedAt: now(), task: task, scheduleBlock: block), in: context)
    }

    func start(habit: HabitRecord, block: ScheduleBlockRecord?, in context: ModelContext) throws {
        try insert(WorkSessionRecord(startedAt: now(), habit: habit, scheduleBlock: block), in: context)
    }

    func pause(in context: ModelContext) throws {
        let session = try requireOpen(in: context)
        guard session.pausedAt == nil else { return }
        session.pausedAt = now()
        session.updatedAt = now()
        try context.save()
    }

    func resume(in context: ModelContext) throws {
        let session = try requireOpen(in: context)
        closePause(session)
        try context.save()
    }

    /// Ends the open session. `markDone` completes its task or logs its
    /// habit for the logical day the session started on.
    func finish(markDone: Bool, in context: ModelContext) throws {
        let session = try requireOpen(in: context)
        closePause(session)
        session.endedAt = now()
        session.updatedAt = now()
        if markDone {
            if let task = session.task {
                task.completedAt = now()
                task.updatedAt = now()
            } else if let habit = session.habit {
                log(habit, workedSeconds: session.snapshot.elapsed(at: now()), on: session.startedAt, in: context)
            }
        }
        try context.save()
    }

    func cancel(in context: ModelContext) throws {
        context.delete(try requireOpen(in: context))
        try context.save()
    }

    // MARK: - Private

    private func insert(_ session: WorkSessionRecord, in context: ModelContext) throws {
        guard try Self.openSession(in: context) == nil else { throw TrackerError.sessionAlreadyOpen }
        context.insert(session)
        try context.save()
    }

    private func requireOpen(in context: ModelContext) throws -> WorkSessionRecord {
        guard let session = try Self.openSession(in: context) else { throw TrackerError.noOpenSession }
        return session
    }

    private func closePause(_ session: WorkSessionRecord) {
        guard let pausedAt = session.pausedAt else { return }
        session.pausedSeconds += max(0, now().timeIntervalSince(pausedAt))
        session.pausedAt = nil
        session.updatedAt = now()
    }

    private func log(_ habit: HabitRecord, workedSeconds: TimeInterval, on day: Date, in context: ModelContext) {
        let instant = boundary.loggingInstant(for: now(), on: boundary.startOfDay(for: day))
        let existing = habit.completions?.first { boundary.isDate($0.date, inSameDayAs: day) }
        switch habit.type {
        case .timer:
            if let existing { existing.value += workedSeconds } else {
                context.insert(CompletionRecord(date: instant, value: workedSeconds, habit: habit))
            }
        case .counter:
            if let existing { existing.value += 1 } else {
                context.insert(CompletionRecord(date: instant, value: 1, habit: habit))
            }
        case .binary, .negative:
            if let existing { existing.value = max(existing.value, 1) } else {
                context.insert(CompletionRecord(date: instant, value: 1, habit: habit))
            }
        }
    }
}
```

```swift
import Foundation
import SwiftData
import KadoCore

/// Reads the store and produces what `NowResolver` and the "Start
/// something…" list need, as values. All "can this be worked on now?"
/// rules live here.
@MainActor
struct NowInputBuilder {
    struct Input: Equatable {
        var blocks: [NowBlock]
        var openSession: OpenSession?
        var startCandidates: [NowItem]
    }

    let boundary: DayBoundary
    let evaluator: any FrequencyEvaluating

    func build(now: Date, in context: ModelContext) throws -> Input {
        let tasks = try context.fetch(FetchDescriptor<TaskRecord>(sortBy: [SortDescriptor(\.title)]))
        let habits = try context.fetch(FetchDescriptor<HabitRecord>(sortBy: [SortDescriptor(\.sortOrder)]))
        let workableTasks = tasks.filter(isWorkable)
        let workableHabits = habits.filter { isWorkable($0, now: now) }
        let taskIDs = Set(workableTasks.map(\.id))
        let habitIDs = Set(workableHabits.map(\.id))

        let blocks = try context.fetch(FetchDescriptor<ScheduleBlockRecord>()).compactMap { block -> NowBlock? in
            guard let start = block.startAt, let item = item(for: block, taskIDs: taskIDs, habitIDs: habitIDs) else { return nil }
            return NowBlock(id: block.id, item: item, start: start, end: block.endAt, createdAt: block.createdAt)
        }

        let open = try WorkSessionTracker.openSession(in: context).flatMap { record -> OpenSession? in
            guard let item = item(task: record.task, habit: record.habit) else { return nil }
            return OpenSession(id: record.id, item: item, session: record.snapshot, blockID: record.scheduleBlock?.id)
        }

        let candidates = workableTasks.map { NowItem.task(id: $0.id, title: $0.title) }
            + workableHabits.map { NowItem.habit(id: $0.id, name: $0.name) }
        return Input(blocks: blocks, openSession: open, startCandidates: candidates)
    }

    private func isWorkable(_ task: TaskRecord) -> Bool {
        task.completedAt == nil && task.archivedAt == nil && task.externalCancelledAt == nil
    }

    private func isWorkable(_ habit: HabitRecord, now: Date) -> Bool {
        guard habit.archivedAt == nil else { return false }
        if case .negative = habit.type { return false }
        let completions = (habit.completions ?? []).compactMap(\.snapshot)
        return evaluator.isOutstanding(habit: habit.snapshot, on: now, completions: completions)
    }

    private func item(for block: ScheduleBlockRecord, taskIDs: Set<UUID>, habitIDs: Set<UUID>) -> NowItem? {
        if let task = block.task, taskIDs.contains(task.id) { return .task(id: task.id, title: task.title) }
        if let habit = block.habit, habitIDs.contains(habit.id) { return .habit(id: habit.id, name: habit.name) }
        return nil
    }

    private func item(task: TaskRecord?, habit: HabitRecord?) -> NowItem? {
        if let task { return .task(id: task.id, title: task.title) }
        if let habit { return .habit(id: habit.id, name: habit.name) }
        return nil
    }
}
```

```swift
            if let plannedRange {
                Text(plannedRange.lowerBound.formatted(date: .omitted, time: .shortened)
                     + "–" + plannedRange.upperBound.formatted(date: .omitted, time: .shortened))
                    .foregroundStyle(.secondary)
            }
            Text("Started \(open.session.startedAt, format: .dateTime.hour().minute())")
            elapsedText
                .font(.title2.monospacedDigit())
                .accessibilityIdentifier(AccessibilityID.Now.elapsed)
            if let plannedRange { progress(plannedRange) }
            buttons
        }
        .accessibilityElement(children: .contain)
    }

    @ViewBuilder
    private var elapsedText: some View {
        let minutes = Int(open.session.elapsed(at: now) / 60)
        if open.session.isPaused {
            Text("Paused · \(minutes) min so far")
        } else {
            // Counts every second without our own timer.
            Text(timerInterval: open.session.startedAt.addingTimeInterval(open.session.pausedSeconds)...Date.distantFuture,
                 countsDown: false)
            + Text(" ") + Text("elapsed")
        }
    }

    private func progress(_ range: ClosedRange<Date>) -> some View {
        let planned = range.upperBound.timeIntervalSince(range.lowerBound)
        let worked = open.session.elapsed(at: now)
        let left = Int((planned - worked) / 60)
        return VStack(alignment: .leading, spacing: 4) {
            ProgressView(value: min(worked, planned), total: max(planned, 1))
            Text(left >= 0 ? "\(left) min left" : "\(-left) min over")
                .font(.footnote).foregroundStyle(.secondary)
        }
    }

    private var buttons: some View {
        ViewThatFits(in: .horizontal) {
            HStack(spacing: 12) { finishButton; pauseOrResume }
            VStack(alignment: .leading, spacing: 12) { finishButton; pauseOrResume }
        }
    }

    private var finishButton: some View {
        Button("Finish", action: onFinish)

            .controlSize(.large)
            .accessibilityIdentifier(AccessibilityID.Now.finish)

                .buttonStyle(.bordered).controlSize(.large)
```

```swift
import Foundation

nonisolated public struct SuggestionCandidate: Hashable, Sendable {
    public var item: NowItem
    public var dueDate: Date?
    public var streakDays: Int

    public init(item: NowItem, dueDate: Date?, streakDays: Int) {
        self.item = item
        self.dueDate = dueDate
        self.streakDays = streakDays
    }
}

nonisolated public enum SuggestionReason: Hashable, Sendable {
    case overdue, dueToday, dueTomorrow, keepStreak(days: Int), habitToday, openTask
}

nonisolated public struct RankedSuggestion: Hashable, Sendable {
    public var candidate: SuggestionCandidate
    public var reason: SuggestionReason

    public init(candidate: SuggestionCandidate, reason: SuggestionReason) {
        self.candidate = candidate
        self.reason = reason
    }
}

/// Orders start candidates for free time. The on-device model (phase 2)
/// only chooses among the first few; this order is also the fallback.
nonisolated public struct SuggestionRanker: Sendable {
    public static let limit = 8
    public let boundary: DayBoundary

    public init(boundary: DayBoundary) {
        self.boundary = boundary
    }

    public func rank(_ candidates: [SuggestionCandidate], now: Date) -> [RankedSuggestion] {
        candidates
            .map { RankedSuggestion(candidate: $0, reason: reason(for: $0, now: now)) }
            .sorted { (priority($0.reason), $0.candidate.item.title) < (priority($1.reason), $1.candidate.item.title) }
            .prefix(Self.limit)
            .map { $0 }
    }

    private func reason(for candidate: SuggestionCandidate, now: Date) -> SuggestionReason {
        switch candidate.item {
        case .habit:
            return candidate.streakDays >= 3 ? .keepStreak(days: candidate.streakDays) : .habitToday
        case .task:
            guard let due = candidate.dueDate else { return .openTask }
            let today = boundary.startOfDay(for: now)
            let dueDay = boundary.startOfDay(for: due)
            if dueDay < today { return .overdue }
            if dueDay == today { return .dueToday }
            if let tomorrow = boundary.calendar.date(byAdding: .day, value: 1, to: today),
               boundary.isDate(due, inSameDayAs: tomorrow) { return .dueTomorrow }
            return .openTask
        }
    }

    private func priority(_ reason: SuggestionReason) -> Int {
        switch reason {
        case .overdue: return 0
        case .dueToday: return 1
        case .dueTomorrow: return 2
        case .keepStreak: return 3
        case .habitToday: return 4
        case .openTask: return 5
        }
    }
}
```

```swift
    @Guide(description: "One short sentence on why, under 15 words, in the user's language")
    var reason: String

/// Lets Apple's on-device model choose among ranked candidates. Only
/// titles, due dates and streaks are sent, and only on device.
@available(iOS 26, *)
struct FoundationModelsNextSuggester: NextSuggesting {
    private let model: SystemLanguageModel
    private let fallback = RankedNextSuggester()

    init(model: SystemLanguageModel = .default) {
        self.model = model
    }

    func suggest(from ranked: [RankedSuggestion], freeMinutes: Int?) async -> NextSuggestion? {
        guard !ranked.isEmpty else { return nil }
        guard case .available = model.availability else {
            return await fallback.suggest(from: ranked, freeMinutes: freeMinutes)
        }
        let session = LanguageModelSession(model: model, instructions: """
            You pick the single best thing for a person to start now, from a list. \
            Prefer what fits the free time, is overdue or due soon, or keeps a habit streak. \
            Answer with one candidate ID from the list and a short reason. Never invent an ID.
            """)
        do {
            let pick = try await session.respond(
                to: Self.prompt(ranked, freeMinutes: freeMinutes),
                generating: ModelPick.self,
                options: GenerationOptions(sampling: .greedy)
            ).content
            return NextSuggestion.validated(candidateID: pick.candidateID, reason: pick.reason, ranked: ranked)
        } catch {
            return await fallback.suggest(from: ranked, freeMinutes: freeMinutes)
        }
    }

    static func prompt(_ ranked: [RankedSuggestion], freeMinutes: Int?) -> String {
        let free = freeMinutes.map { "Free time: \($0) min." } ?? "Free time: rest of the day."
        let lines = ranked.map { entry -> String in
            let c = entry.candidate
            let kind = { if case .task = c.item { return "task" } else { return "habit" } }()
            let due = c.dueDate.map { " | due \($0.formatted(date: .abbreviated, time: .omitted))" } ?? ""
            let streak = c.streakDays > 0 ? " | streak \(c.streakDays) days" : ""
            return "\(c.item.id.uuidString) | \(kind) | \(c.item.title)\(due)\(streak)"
        }
        return ([free, "Candidates:"] + lines).joined(separator: "\n")
    }
```
